{%- comment -%} cite.html — "Cite this" disclosure with copy-to-clipboard buttons for BibTeX, RIS, Hayagriva, and a plain-text citation. Inputs: post — the post to cite (defaults to `page`). The citation copy is rendered server-side as `
` blocks. JS in
  /js/cite.js wires the copy buttons; without JS the user can still
  select-all-and-copy from the visible block, so the affordance
  degrades cleanly.

  Each format embeds:
    · the article's permanent identifier `oc:YYYY/NNNN` from
      `_data/permaids.yml` (assigned at build time, immutable)
    · the OpenTimestamps proof URL when one has been generated
      (presence read from `_data/timestamps.yml`)
    · the canonical article URL, the byline name(s), the publish date,
      and the publication name.
{%- endcomment -%}

{%- assign post = include.post | default: page -%}
{%- assign author = site.authors | where: "slug", post.author | first -%}
{%- assign byline = author.name | default: post.author | default: site.author.name -%}
{%- assign permaid_key = post.permaid_seed | default: post.slug -%}
{%- assign permaid = site.data.permaids[permaid_key] | default: site.data.permaids[post.slug] -%}
{%- assign ts = site.data.timestamps[post.slug] -%}

{%- comment -%}
  Format-specific escapers. We assemble each field once via Liquid
  `capture`/`replace` so the on-page 
 shows exactly the bytes a
  reference manager will ingest.

  · BibTeX field values sit inside `{...}` — we backslash-escape `\`
    and the LaTeX specials `&`, `%`, `#`, `_`, `$`. Literal `{` / `}`
    in a title are not escaped because (a) Liquid 4's parser cannot
    represent them in `replace:` arguments, and (b) editorial titles
    on this publication don't carry curly braces.
  · RIS is line-oriented; embedded newlines must collapse to spaces.
  · Hayagriva is YAML — we double-quote and escape `\` and `"`.
{%- endcomment -%}

{%- capture _bib_title -%}{{ post.title | replace: "\\", "\\\\" | replace: "&", "\\&" | replace: "%", "\\%" | replace: "#", "\\#" | replace: "_", "\\_" | replace: "$", "\\$" }}{%- endcapture -%}
{%- capture _bib_author -%}{{ byline | replace: "\\", "\\\\" | replace: "&", "\\&" }}{%- endcapture -%}

{%- capture _ris_title  -%}{{ post.title | strip_newlines | replace: "  ", " " }}{%- endcapture -%}
{%- capture _ris_author -%}{{ byline    | strip_newlines }}{%- endcapture -%}

{%- capture _yaml_title  -%}{{ post.title | replace: "\\", "\\\\" | replace: '"', '\"' | strip_newlines }}{%- endcapture -%}
{%- capture _yaml_author -%}{{ byline    | replace: "\\", "\\\\" | replace: '"', '\"' | strip_newlines }}{%- endcapture -%}
{%- capture _yaml_pub    -%}{{ site.title | replace: "\\", "\\\\" | replace: '"', '\"' }}{%- endcapture -%}
{%- assign post_url = post.url | absolute_url -%}
{%- assign year = post.date | date: "%Y" -%}
{%- assign iso_date = post.date | date: "%Y-%m-%d" -%}
{%- assign access_date = site.time | date: "%Y-%m-%d" -%}

{%- comment -%} BibTeX key: {first-author-surname}{year}{first-word-of-title} {%- endcomment -%}
{%- assign _surname = byline | split: " " | last | downcase | replace: ".", "" -%}
{%- assign _firstword = post.title | split: " " | first | downcase | replace: ":", "" | replace: ",", "" -%}
{%- capture bib_key -%}{{ _surname }}{{ year }}{{ _firstword }}{%- endcapture -%}

{%- capture bibtex -%}
@misc{ {{- bib_key -}},
  author       = { {{- _bib_author -}} },
  title        = { {{- _bib_title -}} },
  howpublished = { {{- site.title }}, {{ post_url -}} },
  year         = { {{- year -}} },
  month        = { {{- post.date | date: "%b" | downcase -}} },
  day          = { {{- post.date | date: "%-d" -}} },
  url          = { {{- post_url -}} },
  urldate      = { {{- access_date -}} },
  note         = { {{- permaid }}{% if ts.ots_present %}; OpenTimestamps proof: {{ '/timestamps/' | append: post.slug | append: '.ots' | absolute_url }}{% endif -%} }
}
{%- endcapture -%}
